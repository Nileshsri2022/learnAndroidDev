# Adding The Notifications Part 1

---

## Overview

When a member is added to a board, the app sends an **HTTP POST** to the FCM endpoint with the new member's token, title and message. (Demo: user "Frank" adds the emulator user → notification "Assigned to the board …".)

---

## Constants

```kotlin
const val FCM_BASE_URL: String = "https://fcm.googleapis.com/fcm/send"
const val FCM_AUTHORIZATION: String = "authorization"
const val FCM_KEY: String = "key"
const val FCM_SERVER_KEY: String = "YOUR_SERVER_KEY"   // Project settings → Cloud Messaging
const val FCM_KEY_TITLE: String = "title"
const val FCM_KEY_MESSAGE: String = "message"
const val FCM_KEY_DATA: String = "data"
const val FCM_KEY_TO: String = "to"
```

---

## Sending (MembersActivity)

```kotlin
private inner class SendNotificationToUserAsyncTask(val boardName: String, val token: String)
    : AsyncTask<Any, Void, String>() {

    override fun onPreExecute() {
        super.onPreExecute()
        showProgressDialog(resources.getString(R.string.please_wait))
    }

    override fun doInBackground(vararg params: Any): String {
        var result: String
        var connection: HttpURLConnection? = null
        try {
            val url = URL(Constants.FCM_BASE_URL)
            connection = url.openConnection() as HttpURLConnection
            connection.doOutput = true
            connection.doInput = true
            connection.instanceFollowRedirects = false
            connection.requestMethod = "POST"
            connection.setRequestProperty("Content-Type", "application/json")
            connection.setRequestProperty("charset", "utf-8")
            connection.setRequestProperty("Accept", "application/json")
            connection.setRequestProperty(Constants.FCM_AUTHORIZATION,
                "${Constants.FCM_KEY}=${Constants.FCM_SERVER_KEY}")
            connection.useCaches = false

            val wr = DataOutputStream(connection.outputStream)
            val jsonRequest = JSONObject()
            val dataObject = JSONObject()
            dataObject.put(Constants.FCM_KEY_TITLE, "Assigned to the Board $boardName")
            dataObject.put(Constants.FCM_KEY_MESSAGE,
                "You have been assigned to the new board by ${mAssignedMembersList[0].name}")
            jsonRequest.put(Constants.FCM_KEY_DATA, dataObject)
            jsonRequest.put(Constants.FCM_KEY_TO, token)

            wr.writeBytes(jsonRequest.toString())
            wr.flush(); wr.close()

            val httpResult: Int = connection.responseCode
            result = if (httpResult == HttpURLConnection.HTTP_OK) {
                connection.inputStream.bufferedReader().readText()
            } else {
                connection.responseMessage
            }
        } catch (e: SocketTimeoutException) {
            result = "Connection Timeout"
        } catch (e: Exception) {
            result = "Error : " + e.message
        } finally {
            connection?.disconnect()
        }
        return result
    }

    override fun onPostExecute(result: String) {
        super.onPostExecute(result)
        hideProgressDialog()
        Log.e("JSON Response Result", result)
    }
}

fun memberAssignSuccess(user: User) {
    …
    SendNotificationToUserAsyncTask(mBoardDetails.name, user.fcmToken).execute()
}
```

Reuses the POST technique from the Weather chapter demo.

> ⚠️ Embedding a server key in an app is insecure, and the **legacy FCM HTTP API has been shut down (2024)**. Today, send notifications from a trusted backend (e.g. Cloud Functions) using the **FCM HTTP v1 API** with OAuth.

---

## Recap / cheat sheet

- POST `{ "to": token, "data": { title, message } }` to FCM with `authorization: key=…`.
- Trigger after `memberAssignSuccess`.
