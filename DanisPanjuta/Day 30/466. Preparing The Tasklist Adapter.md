# Preparing The Tasklist Adapter

---

## Overview

Defines the **`Task`** model (a list/column) and **`TaskListItemsAdapter`**. Each item is a column ~70% of screen width; the **last item** is always an "**Add List**" button.

## Adapter essentials

```kotlin
open class TaskListItemsAdapter(
    private val context: Context,
    private var list: ArrayList<Task>
) : RecyclerView.Adapter<RecyclerView.ViewHolder>() {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val view = LayoutInflater.from(context).inflate(R.layout.item_task, parent, false)
        val layoutParams = LinearLayout.LayoutParams(
            (parent.width * 0.7).toInt(), LinearLayout.LayoutParams.WRAP_CONTENT)
        layoutParams.setMargins((15.toDp()).toPx(), 0, (40.toDp()).toPx(), 0)
        view.layoutParams = layoutParams
        return MyViewHolder(view)
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        val model = list[position]
        if (holder is MyViewHolder) {
            if (position == list.size - 1) {
                holder.itemView.tv_add_task_list.visibility = View.VISIBLE
                holder.itemView.ll_task_item.visibility = View.GONE
            } else {
                holder.itemView.tv_add_task_list.visibility = View.GONE
                holder.itemView.ll_task_item.visibility = View.VISIBLE
            }
            holder.itemView.tv_task_list_title.text = model.title
        }
    }

    override fun getItemCount() = list.size

    private fun Int.toDp(): Int = (this / Resources.getSystem().displayMetrics.density).toInt()
    private fun Int.toPx(): Int = (this * Resources.getSystem().displayMetrics.density).toInt()

    class MyViewHolder(view: View) : RecyclerView.ViewHolder(view)
}
```

- **Extension functions** `Int.toDp()` / `Int.toPx()` add methods to `Int`.
- 70% width lets the next column peek in — hinting horizontal scroll.

## Activity

```kotlin
fun boardDetails(board: Board) {
    …
    val addTaskList = Task(resources.getString(R.string.add_list))
    board.taskList.add(addTaskList)                     // placeholder for "Add List"

    rv_task_list.layoutManager =
        LinearLayoutManager(this, LinearLayoutManager.HORIZONTAL, false)
    rv_task_list.setHasFixedSize(true)
    rv_task_list.adapter = TaskListItemsAdapter(this, board.taskList)
}
```

---

## Recap / cheat sheet

- Horizontal RecyclerView of `Task` columns; last item = "Add List".
- Dynamic item width via `LayoutParams` in `onCreateViewHolder`.
