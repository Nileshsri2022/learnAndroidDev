## 📝 Summary

In this capstone lecture of the Firebase Chat application, students construct the primary conversational interface: the **`ChatScreen`**. Powered by `MessageViewModel`, this screen connects the real-time `callbackFlow` stream from `MessageRepository` directly to a high-performance `LazyColumn` message thread.

Denis demonstrates how to capture text input in a bottom action bar, dispatch outgoing messages via `viewModel.sendMessage()`, format UTC epoch timestamps into human-readable hours and minutes, and dynamically style message bubbles—rendering the current user's messages on the right in purple, and other participants' messages on the left in subtle gray.

## 📒 Detailed Code Implementation

### 1. The Message ViewModel (`MessageViewModel.kt`)
```kotlin
package eu.tutorials.chatroomapp.viewmodel

import androidx.lifecycle.*
import eu.tutorials.chatroomapp.Injection
import eu.tutorials.chatroomapp.data.*
import kotlinx.coroutines.launch

class MessageViewModel : ViewModel() {

    private val messageRepository: MessageRepository = MessageRepository(Injection.instance())
    private val userRepository: UserRepository = UserRepository(
        com.google.firebase.auth.FirebaseAuth.getInstance(),
        Injection.instance()
    )

    private val _messages = MutableLiveData<List<Message>>()
    val messages: LiveData<List<Message>> get() = _messages

    private val _roomId = MutableLiveData<String>()
    private var currentUser: User? = null

    init {
        // Fetch current user profile to determine bubble ownership
        viewModelScope.launch {
            when (val userResult = userRepository.getCurrentUser()) {
                is Result.Success -> currentUser = userResult.data
                is Result.Error -> { /* Handle error */ }
            }
        }
    }

    // Set active room and collect real-time message stream
    fun setRoomId(roomId: String) {
        _roomId.value = roomId
        viewModelScope.launch {
            messageRepository.getChatMessages(roomId).collect { messageList ->
                // Flag messages sent by current user for UI styling
                val updatedList = messageList.map { message ->
                    message.copy(isSentByCurrentUser = message.senderId == currentUser?.email)
                }
                _messages.value = updatedList
            }
        }
    }

    // Send a new message
    fun sendMessage(text: String) {
        val user = currentUser ?: return
        val roomId = _roomId.value ?: return

        if (text.isNotBlank()) {
            val message = Message(
                senderFirstName = user.firstName,
                senderId = user.email,
                text = text.trim(),
                timestamp = System.currentTimeMillis()
            )
            viewModelScope.launch {
                messageRepository.sendMessage(roomId, message)
            }
        }
    }
}
```

### 2. The Chat Screen Composable (`chat.kt`)
```kotlin
package eu.tutorials.chatroomapp.screen

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.automirrored.filled.Send
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.runtime.livedata.observeAsState
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import androidx.lifecycle.viewmodel.compose.viewModel
import eu.tutorials.chatroomapp.data.Message
import eu.tutorials.chatroomapp.viewmodel.MessageViewModel
import java.text.SimpleDateFormat
import java.util.*

@Composable
fun ChatScreen(
    roomId: String,
    roomName: String,
    messageViewModel: MessageViewModel = viewModel()
) {
    var text by remember { mutableStateOf("") }
    val messages by messageViewModel.messages.observeAsState(emptyList())

    // Initialize room listener when screen loads
    LaunchedEffect(roomId) {
        messageViewModel.setRoomId(roomId)
    }

    Column(modifier = Modifier.fillMaxSize()) {
        // Chat Header
        Surface(shadowElevation = 4.dp, modifier = Modifier.fillMaxWidth()) {
            Text(
                text = roomName,
                fontSize = 20.sp,
                fontWeight = FontWeight.Bold,
                modifier = Modifier.padding(16.dp)
            )
        }

        // Live Message Feed
        LazyColumn(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 8.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            items(messages) { message ->
                ChatMessageItem(message = message)
            }
        }

        // Message Input Row
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(8.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            OutlinedTextField(
                value = text,
                onValueChange = { text = it },
                placeholder = { Text("Type a message...") },
                modifier = Modifier.weight(1f),
                singleLine = true
            )
            Spacer(modifier = Modifier.width(8.dp))
            IconButton(
                onClick = {
                    if (text.isNotBlank()) {
                        messageViewModel.sendMessage(text)
                        text = "" // Reset input after send
                    }
                }
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.Send,
                    contentDescription = "Send Message",
                    tint = MaterialTheme.colorScheme.primary
                )
            }
        }
    }
}

@Composable
fun ChatMessageItem(message: Message) {
    val isCurrentUser = message.isSentByCurrentUser
    val bubbleColor = if (isCurrentUser) MaterialTheme.colorScheme.primary else Color(0xFFE0E0E0)
    val textColor = if (isCurrentUser) Color.White else Color.Black
    val alignment = if (isCurrentUser) Alignment.End else Alignment.Start

    Column(
        modifier = Modifier.fillMaxWidth(),
        horizontalAlignment = alignment
    ) {
        // Author Name & Time Header
        val formattedTime = SimpleDateFormat("h:mm a", Locale.getDefault())
            .format(Date(message.timestamp))
        Text(
            text = "${if (isCurrentUser) "You" else message.senderFirstName} · $formattedTime",
            fontSize = 11.sp,
            color = Color.Gray,
            modifier = Modifier.padding(bottom = 2.dp)
        )

        // Bubble Container
        Box(
            modifier = Modifier
                .background(
                    color = bubbleColor,
                    shape = RoundedCornerShape(
                        topStart = 16.dp,
                        topEnd = 16.dp,
                        bottomStart = if (isCurrentUser) 16.dp else 0.dp,
                        bottomEnd = if (isCurrentUser) 0.dp else 16.dp
                    )
                )
                .padding(horizontal = 14.dp, vertical = 10.dp)
        ) {
            Text(text = message.text, color = textColor, fontSize = 15.sp)
        }
    }
}
```

## 📌 Important Terms

| Term | Definition |
| :--- | :--- |
| **`Flow.collect`** | Suspending terminal operator in Kotlin Coroutines that receives emitted stream items sequentially. |
| **Message Bubble** | Visual UI pattern displaying chat text enclosed in rounded, colored containers with directional tails. |
| **Epoch Milliseconds** | UTC time represented as the number of milliseconds elapsed since January 1, 1970. |
| **`SimpleDateFormat`** | Android utility for formatting epoch timestamps into localized strings (e.g. `"h:mm a"`). |

## ❓ Q&A

**Q: Why do we reset `text = ""` immediately upon sending?**  
A: Resetting the input state gives the user instant visual confirmation that their tap registered. Because Firestore's write is asynchronous, wiping the input field prevents accidental double-sends.

**Q: Can messages be deleted or edited?**  
A: Yes! You can add delete or update functionality by invoking `firestore.collection("rooms").document(roomId).collection("messages").document(messageId).delete()` or `.update("text", newText)`.
