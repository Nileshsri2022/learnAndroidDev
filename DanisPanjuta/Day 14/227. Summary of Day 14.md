# Summary of Day 14

The course's **Summary of Day 14** lecture has no transcript. This recap consolidates the full Room architecture and CRUD flow built throughout Day 14.

## Theme

Persist a wishlist: every wish added via the form is written to a **Room SQLite file** and re-read as a `Flow<List<Wish>>` that recomposes the list. The app survives kills and reboots.

---

## Architectural flow

```text
Application.onCreate → Graph.provide(this)
                           ├── database = Room.databaseBuilder(appContext, WishDatabase::class.java, "wishlist.db").build()
                           └── wishRepository = WishRepository(database.wishDao())

WishViewModel(Graph.wishRepository)
      ├── getAllWishes: Flow<List<Wish>> = wishRepository.getWishes()  (lateinit)
      ├── addWish(Wish)    → viewModelScope(IO) → repository.addWish  → Dao INSERT
      ├── getWishById(id): Flow<Wish>               → Dao SELECT WHERE id
      ├── updateWish(Wish) → IO → repository.update → Dao UPDATE
      └── deleteWish(Wish) → IO → repository.delete → Dao DELETE

HomeView
      ├── val wishlist by viewModel.getAllWishes.collectAsState(emptyList())
      ├── LazyColumn { items(wishlist, key={id}) {
      │        SwipeToDismiss(rememberDismissState{ deleteWish }, EndToStart, background=Box(Red+Delete))
      │             → WishItem(wish) → navigate(AddScreen/id)
      │   }
      └── Scaffold(add Snackbars)

AddEditDetailView(id, viewModel, navController)
      ├── id→ getWishById.collectAsState → prefill WishTextField
      └── Button: if(update) viewModel.updateWish(Wish(id,...)) else addWish(Wish(...)) → navigateUp

WishDao (@Dao)
      ├── @Insert suspend addWish
      ├── @Query("SELECT * FROM wishlist") Flow<List<Wish>> getAllWishes
      ├── @Update suspend updateWish
      ├── @Delete suspend deleteWish
      └── @Query("SELECT * FROM wishlist WHERE id=:id") Flow<Wish> getWishById

WishDatabase (@Database(entities=[Wish], version=1))
      └── abstract fun wishDao(): WishDao  (Impl generated via KAPT)

Wish (@Entity(tableName="wishlist"))
      └── @PrimaryKey(autoGenerate=true) Long id, @ColumnInfo title/description
```

## Complete reference implementation (Room portion)

### Wish.kt
```kotlin
@Entity(tableName="wishlist")
data class Wish(@PrimaryKey(autoGenerate=true) val id: Long=0L,
                @ColumnInfo(name="wish-title") val title:String="",
                @ColumnInfo(name="wish-desc") val description:String="")
```

### WishDao.kt
```kotlin
@Dao interface WishDao {
    @Insert(onConflict=IGNORE) suspend fun addWish(w:Wish)
    @Query("SELECT * FROM `wishlist`") fun getAllWishes(): Flow<List<Wish>>
    @Update suspend fun updateWish(w:Wish)
    @Delete suspend fun deleteWish(w:Wish)
    @Query("SELECT * FROM `wishlist` WHERE id=:id") fun getWishById(id:Long): Flow<Wish>
}
```

### WishDatabase.kt
```kotlin
@Database(entities=[Wish::class], version=1, exportSchema=false)
abstract class WishDatabase:RoomDatabase(){ abstract fun wishDao():WishDao }
```

### WishRepository.kt
```kotlin
class WishRepository(private val wDao:WishDao){
    suspend fun addWish(w:Wish)=wDao.addWish(w)
    fun getWishes():Flow<List<Wish>>=wDao.getAllWishes()
    fun getWishById(id:Long)=wDao.getWishById(id)
    suspend fun updateWish(w:Wish)=wDao.updateWish(w)
    suspend fun deleteWish(w:Wish)=wDao.deleteWish(w)
}
```

### Graph.kt
```kotlin
object Graph{
    lateinit var database:WishDatabase
    val wishRepository by lazy{ WishRepository(database.wishDao()) }
    fun provide(context:Context){ database=Room.databaseBuilder(context, WishDatabase::class.java,"wishlist.db").build() }
}
```

### WishlistApp.kt
```kotlin
class WishlistApp:Application(){ override fun onCreate(){ super.onCreate(); Graph.provide(this) } }
// + Manifest: <application android:name=".WishlistApp">
```
