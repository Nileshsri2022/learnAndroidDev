# Finalizing the BMI Calculator with the us system

---

## Overview

Implements the **US-unit formula** and makes the CALCULATE button branch on the currently visible view.

---

## Formula

```
BMI = 703 × weight(lb) / (height(in))²
height(in) = feet × 12 + inches
```

---

## Code

```kotlin
binding?.btnCalculateUnits?.setOnClickListener { calculateUnits() }

private fun calculateUnits() {
    if (currentVisibleView == METRIC_UNITS_VIEW) {
        if (validateMetricUnits()) {
            val heightValue = binding?.etMetricUnitHeight?.text.toString().toFloat() / 100
            val weightValue = binding?.etMetricUnitWeight?.text.toString().toFloat()
            displayBMIResult(weightValue / (heightValue * heightValue))
        } else {
            Toast.makeText(this, "Please enter valid values.", Toast.LENGTH_SHORT).show()
        }
    } else {
        if (validateUsUnits()) {
            val usUnitHeightValueFeet = binding?.etUsUnitHeightFeet?.text.toString()
            val usUnitHeightValueInch = binding?.etUsUnitHeightInch?.text.toString()
            val usUnitWeightValue = binding?.etUsUnitWeight?.text.toString().toFloat()

            val heightValue = usUnitHeightValueInch.toFloat() + usUnitHeightValueFeet.toFloat() * 12
            val bmi = 703 * (usUnitWeightValue / (heightValue * heightValue))
            displayBMIResult(bmi)
        } else {
            Toast.makeText(this, "Please enter valid values.", Toast.LENGTH_SHORT).show()
        }
    }
}

private fun validateUsUnits(): Boolean {
    var isValid = true
    when {
        binding?.etUsUnitWeight?.text.toString().isEmpty() -> isValid = false
        binding?.etUsUnitHeightFeet?.text.toString().isEmpty() -> isValid = false
        binding?.etUsUnitHeightInch?.text.toString().isEmpty() -> isValid = false
    }
    return isValid
}
```

Example: 160 lb, 5 ft 9 in → 69 in → 703 × 160 / 4761 ≈ **23.6** (Normal).

---

## Recap / cheat sheet

- **US:** `703 * lb / in²`, inches = `ft*12 + in`.
- Branch on `currentVisibleView`.
